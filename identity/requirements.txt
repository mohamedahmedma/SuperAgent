# Verifying legacy bcrypt hashes imported from the old chat backend. Only ever read,
# never written — a verified legacy password is immediately re-hashed to PBKDF2, so
# this dependency drains away as users sign in and can eventually be dropped.
#
# The bcrypt library directly, NOT passlib. passlib 1.7.4 raises on bcrypt 5.x before
# it reaches the comparison, so every legacy account would fail to verify — which is
# what the old backend was silently doing, since it caught the error and returned
# False. See identity/auth.py.
bcrypt>=4.0.0
